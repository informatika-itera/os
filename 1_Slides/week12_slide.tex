\section{Konsep Persistence \& Abstraksi File}

%--- Slide 1: Masalah Volatilitas ---
\begin{frame}[shrink=5]
    \frametitle{Masalah: RAM itu Pelupa}
    \framesubtitle{Data hilang saat listrik mati atau program selesai}

    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{alertblock}{Masalah Volatilitas}
            \begin{itemize}
                \item Memori utama (RAM) bersifat \tem{volatile}.
                \item Semua data, perhitungan, dan \textit{state} proses hilang saat:
                \begin{itemize}
                    \item Komputer dimatikan/crash.
                    \item Program (proses) selesai jalan.
                \end{itemize}
            \end{itemize}
        \end{alertblock}

        \column{0.48\textwidth}
        \begin{exampleblock}{Kebutuhan Kita}
            Kita butuh menyimpan data hasil kerja (dokumen, foto, kode) agar tetap ada esok hari.
            Kebutuhan ini menuntut ruang penyimpanan yang besar, tahan lama, dan aman.
        \end{exampleblock}
    \end{columns}
\end{frame}

%--- Slide 2: Analogi ---
\begin{frame}
    \frametitle{Analogi: Meja Kerja vs Lemari Arsip}
    \framesubtitle{Mengapa kita butuh dua jenis tempat penyimpanan}

    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{RAM = Meja Kerja}
            \begin{itemize}
                \item Sangat dekat dan cepat dijangkau.
                \item Kapasitas terbatas.
                \item Di akhir hari (dimatikan), meja harus dibersihkan total.
            \end{itemize}
        \end{block}

        \column{0.48\textwidth}
        \begin{block}{Storage = Lemari Arsip Baja}
            \begin{itemize}
                \item Sedikit lebih jauh dan lambat.
                \item Kapasitas luar biasa besar.
                \item Data \tem{aman dan permanen} meskipun kantor sedang tutup.
            \end{itemize}
        \end{block}
    \end{columns}

    \vspace{0.4cm}
    \begin{exampleblock}{Tugas Sistem Operasi}
        OS bertugas memindahkan data dari "meja kerja" ke dalam "lemari arsip" secara rapi, agar mudah dicari lagi besok.
    \end{exampleblock}
\end{frame}

%--- Slide 3: Persistence ---
\begin{frame}
    \frametitle{Konsep Persistence (Ketahanan Data)}
    \framesubtitle{Menjaga data tetap hidup melampaui proses yang membuatnya}
    \small

    \begin{columns}[T]
        \column{0.50\textwidth}
        \textbf{Apa itu Persistence?}
        \begin{itemize}
            \item Kemampuan sistem mempertahankan \textit{state} (data) meskipun listrik terputus.
            \item Menggunakan \textit{hardware} khusus: \textit{Hard Disk Drive} (HDD) atau \textit{Solid State Drive} (SSD).
        \end{itemize}
        
        \column{0.46\textwidth}
        \textbf{Tantangan OS:}
        \begin{itemize}
            \item Disk fisik hanyalah deretan panjang sektor-sektor mentah (biasanya 512 bytes).
            \item Bagaimana cara pengguna membedakan "foto liburan" dengan "tugas OS" di disk yang sama?
        \end{itemize}
    \end{columns}

    \vspace{0.4cm}
    \begin{alertblock}{Solusi: Abstraksi}
        OS tidak membiarkan pengguna membaca disk per sektor. OS menciptakan abstraksi yang mudah dipahami: \tem{File} dan \tem{Direktori}.
    \end{alertblock}
\end{frame}

%--- Slide 4: Abstraksi File ---
\begin{frame}
    \frametitle{Abstraksi Pertama: File}
    \framesubtitle{Hanya sekumpulan byte dengan nama}
    \small

    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{Apa itu File?}
            File adalah abstraksi OS untuk kumpulan data yang saling terkait. Bagi OS, file hanyalah sebuah \tem{\textit{array of bytes}} linier.
        \end{block}

        \begin{itemize}
            \item Setiap file punya \textbf{nama tingkat tinggi} (\texttt{tugas.pdf}).
            \item OS yang bertugas memetakan nama file ini ke blok/sektor fisik di dalam disk.
        \end{itemize}

        \column{0.48\textwidth}
        \begin{exampleblock}{Nomor Inode}
            Di sistem UNIX/Linux, di bawah layar, setiap file sebenarnya diidentifikasi dengan nomor unik yang disebut \tem{\textit{inode number}}. OS mengubah nama file menjadi nomor ini.
        \end{exampleblock}
    \end{columns}
\end{frame}

%--- Slide 5: Operasi Dasar File ---
\begin{frame}[fragile]
    \frametitle{Mekanisme Interaksi dengan File}
    \framesubtitle{Sistem operasi menyediakan API standar untuk membaca/menulis}
    \footnotesize

    \begin{columns}[T]
        \column{0.54\textwidth}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
// 1. Buka/buat file
int fd = open("tugas.txt", O_CREAT | O_WRONLY);

// 2. Tulis data
write(fd, "Halo dunia", 10);

// 3. Sinkronisasi (opsional)
fsync(fd);

// 4. Tutup
close(fd);
\end{lstlisting}

        \column{0.42\textwidth}
        \begin{block}{API File}
            \begin{enumerate}
                \item \texttt{open()}: Minta izin OS, dapat \textit{file descriptor}.
                \item \texttt{read() / write()}: Pindahkan data.
                \item \texttt{fsync()}: Paksa data turun dari cache RAM langsung ke disk fisik.
                \item \texttt{close()}: Lepas sumber daya.
            \end{enumerate}
        \end{block}
    \end{columns}
\end{frame}

%--- Slide 6: Abstraksi Direktori ---
\begin{frame}
    \frametitle{Abstraksi Kedua: Direktori}
    \framesubtitle{Agar ribuan file tidak berserakan}
    \small

    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{Apa itu Direktori?}
            Secara teknis, direktori hanyalah \tem{file khusus} yang berisi daftar pasangan: \\
            \texttt{(Nama File $\to$ Nomor Inode)}
        \end{block}
        
        \begin{itemize}
            \item Karena direktori bisa menyimpan direktori lain, terbentuklah struktur \tem{hierarki (\textit{tree})}.
            \item Berawal dari satu \textit{root} (\texttt{/}).
        \end{itemize}

        \column{0.48\textwidth}
        \begin{exampleblock}{Analogi}
            File adalah "dokumen" lembaran. Direktori adalah "map" (folder) yang mengelompokkan dokumen-dokumen terkait dalam laci lemari arsip.
        \end{exampleblock}
    \end{columns}
\end{frame}

%--- Slide 6: Checkpoint 1 ---
\begin{frame}
    \frametitle{Checkpoint 1: Persistence \& Abstraksi}
    \framesubtitle{Cek pemahaman sebelum lanjut}
    \small
    \begin{enumerate}
        \item Mengapa kita tidak menyimpan semua data secara permanen di RAM saja?
        \item Apa bedanya cara pengguna melihat file vs cara OS melihat file?
        \item Apa isi dari sebuah direktori pada tingkatan teknis terbawah?
    \end{enumerate}
    \vspace{0.5cm}
    \begin{block}{Kunci Jawaban Singkat}
        1. RAM bersifat volatile (hilang saat mati) dan kapasitasnya mahal/terbatas.\\
        2. Pengguna melihat nama file (tingkat tinggi), OS melihat file sebagai \textit{array of bytes} yang dipetakan ke blok fisik disk lewat inode.\\
        3. Daftar pemetaan antara "Nama File" ke "Nomor Inode".
    \end{block}
\end{frame}

\section{Struktur Internal: Inode \& Metadata}

\begin{frame}[shrink=5]

    \framesubtitle{Pemisahan antara nama, metadata, dan data}
    \small

    \begin{columns}[T]
        \column{0.50\textwidth}
        Bagi sistem operasi, file \textbf{bukanlah} namanya.\\
        File direpresentasikan oleh sebuah struktur data bernama \tem{Inode}.
        \vspace{0.3cm}

        \textbf{Tiga elemen utama file:}
        \begin{enumerate}
            \item \textbf{Nama}: Disimpan di Direktori.
            \item \textbf{Metadata}: Disimpan di Inode.
            \item \textbf{Isi}: Disimpan di \textit{Data Blocks} (Disk).
        \end{enumerate}

%--- Slide 2: Analogi Inode ---
\begin{frame}
    \frametitle{Analogi: Perpustakaan Tradisional}
    \framesubtitle{Inode seperti kartu katalog}
    \small

    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{Kartu Katalog = Inode}
            \begin{itemize}
                \item Berisi informasi: Judul buku, pengarang, tahun terbit, tebal buku.
                \item \textbf{Lokasi fisik}: Rak A, Baris 2.
                \item Kartunya sendiri \tem{bukan buku}, ukurannya kecil dan seragam.
            \end{itemize}
        \end{block}
        
        \column{0.48\textwidth}
        \begin{block}{Buku Fisik = Data Block}
            \begin{itemize}
                \item Buku asli yang tebal dan memakan banyak tempat.
                \item Ada di rak penyimpanan utama (disk fisik).
            \end{itemize}
        \end{block}
    \end{columns}

    \vspace{0.4cm}
    \begin{exampleblock}{Kenapa dipisah?}
        Sangat efisien mencari kartu katalog yang rapi di laci depan daripada harus membongkar seluruh rak gudang hanya untuk mencari siapa pengarang buku.
    \end{exampleblock}
\end{frame}

%--- Slide 3: Melihat Inode (Unix Command) ---
\begin{frame}[fragile]
    \frametitle{Melihat Inode di Terminal}
    \framesubtitle{Menggunakan perintah \texttt{ls -i} dan \texttt{ls -l}}
    \small

    \begin{columns}[T]
        \column{0.55\textwidth}
        \begin{block}{Menampilkan Nomor Inode (\texttt{ls -i})}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
$ ls -i
131093 OS_Slide.pdf
131094 tugas_akhir.docx
\end{lstlisting}
        \end{block}

        \begin{block}{Menampilkan Metadata Dasar (\texttt{ls -l})}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
$ ls -l OS_Slide.pdf
-rw-r--r-- 1 mario users 1048576 May 5 10:00 OS_Slide.pdf
\end{lstlisting}
        \end{block}

        \column{0.42\textwidth}
        \textbf{Apa artinya?}
        \begin{itemize}
            \item \texttt{131093}: Inode number.
            \item \texttt{-rw-r--r--}: Permissions.
            \item \texttt{1}: Jumlah \textit{hard link}.
            \item \texttt{mario users}: Owner \& Group.
            \item \texttt{1048576}: Size (dalam bytes).
            \item \texttt{May 5}: Last modified.
        \end{itemize}
    \end{columns}
\end{frame}

    \end{columns}
\end{frame}

%--- Slide 6: Indirect Pointers ---
\begin{frame}[shrink=10]

    \framesubtitle{Single, Double, dan Triple Indirect Pointers}
    \small

    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{exampleblock}{Solusi: Blok Indeks}
            OS menggunakan \tem{Indirect Pointer}: pointer di inode \textbf{tidak} menunjuk ke data, melainkan menunjuk ke sebuah blok penuh berisi pointer lain.

%--- Slide 7: Hard link vs Soft Link ---
\begin{frame}[fragile,shrink=5]
    \frametitle{Hard Link vs Soft (Symbolic) Link}
    \framesubtitle{Dua cara memberikan nama ganda pada sebuah file}
    \small

    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{Hard Link (\texttt{ln})}
        \begin{itemize}
            \item Membuat \textbf{nama baru} di direktori yang menunjuk ke \tem{Inode yang sama}.
            \item Jika file asli dihapus, datanya tetap ada selama masih ada nama lain yang menunjuk (Link count $> 0$).
        \end{itemize}

        \vspace{0.2cm}
        \centering
        \begin{tikzpicture}[scale=0.7, transform shape]
            \node[draw, fill=yellow!20, font=\scriptsize] (d1) at (0,1) {"A.txt"};
            \node[draw, fill=yellow!20, font=\scriptsize] (d2) at (0,0) {"B.txt"};
            \node[draw, fill=blue!20, font=\scriptsize, align=center] (inode) at (3,0.5) {Inode \#10\\Link: 2};
            
            \draw[->, thick] (d1.east) -- (inode.west);
            \draw[->, thick] (d2.east) -- (inode.west);
        \end{tikzpicture}

        \column{0.48\textwidth}
        \textbf{Soft Link (\texttt{ln -s})}
        \begin{itemize}
            \item Memiliki \tem{Inode sendiri}.
            \item Berisi teks berupa \textit{path} string ke file asli (seperti \textit{shortcut} di Windows).
            \item Jika file asli dihapus, link ini menjadi "mati" (\textit{dangling link}).
        \end{itemize}

        \vspace{0.2cm}

%--- Slide 8: Path Traversal ---
\begin{frame}[shrink=10]
    \frametitle{Path Traversal}
    \framesubtitle{Cara OS mencari \texttt{/home/os.pdf}}
    \small

    Untuk membuka sebuah file, OS harus membaca rentetan direktori dan inode secara bergantian.\\
    OS selalu mulai dari \tem{Root Inode} (biasanya bernomor 2).

    \vspace{0.4cm}
    \centering

\section{Penjadwalan Disk (Disk Scheduling)}

%--- Slide 20: Mengapa Butuh Penjadwalan Disk? ---
\begin{frame}[shrink=5]
    \frametitle{Mengapa Butuh Penjadwalan Disk?}
    \framesubtitle{Masalah utama pada komponen mekanis}
    \small
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{I/O Disk Sangat Lambat}
            Tidak seperti RAM yang elektronis, HDD (Hard Disk Drive) memiliki komponen \tem{mekanis} yang bergerak fisik. CPU bisa mengeksekusi jutaan instruksi dalam waktu yang dibutuhkan HDD untuk membaca satu blok data.
        \end{block}
        \textbf{Tujuan Penjadwalan:}
        \begin{itemize}
            \item Mengurangi total waktu pergerakan fisik disk.
            \item Memaksimalkan \textit{throughput} (jumlah request I/O per detik).
        \end{itemize}

        \column{0.48\textwidth}
        \begin{exampleblock}{Analogi: Kurir Gudang}
            Bayangkan kurir di gudang raksasa. Jika dia mengambil paket berdasarkan urutan pesanan masuk tanpa memikirkan rute (loncat dari rak ujung kiri ke ujung kanan terus menerus), dia akan kelelahan dan lambat. \tem{Rute harus dioptimasi.}
        \end{exampleblock}
    \end{columns}
\end{frame}

%--- Slide 21: Anatomi Sederhana Hard Disk ---
\begin{frame}
    \frametitle{Anatomi Hard Disk \& Komponen Waktu}
    \framesubtitle{Dari mana kelambatan itu berasal?}
    \small
    \begin{columns}[T]
        \column{0.50\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.8, transform shape]
            % Disk platter
            \draw[fill=gray!10, thick] (0,0) circle (2.5);
            \draw[fill=gray!30, thick] (0,0) circle (1.8);
            \draw[fill=gray!50, thick] (0,0) circle (1.1);
            \draw[fill=black] (0,0) circle (0.1);
            
            % Tracks labels

            \node at (0, -1.45) {\scriptsize Track dalam};

            % Actuator arm
            \draw[fill=blue!20, thick, rounded corners=2pt] (3, -2) -- (1.5, 0) -- (1.7, 0.2) -- (3.2, -1.8) -- cycle;
            \draw[fill=red!50] (1.5,0) circle (0.1) node[above left] {\scriptsize Head};
            \draw[fill=black] (3.1, -1.9) circle (0.15) node[below] {\scriptsize Pivot};

            \draw[<->, thick, red] (1.7, 0.3) -- (2.3, 0.9) node[midway, right] {\scriptsize Seek};

        \end{tikzpicture}
        \end{center}

        \column{0.48\textwidth}
        \begin{block}{Dua Faktor Penentu Waktu}
            \begin{enumerate}
                \item \tem{Seek Time}: Waktu yang dibutuhkan lengan (\textit{actuator arm}) untuk memindahkan \textit{head} ke \textit{track/silinder} yang tepat. \textit{(Sangat lambat)}
                \item \tem{Rotational Latency}: Waktu tunggu hingga \textit{sektor} yang dicari berputar tepat di bawah \textit{head}. \textit{(Cukup lambat)}
            \end{enumerate}
        \end{block}
        \textit{Catatan: Disk Scheduling modern fokus meminimalkan \textbf{Seek Time}.}
    \end{columns}
\end{frame}

%--- Slide 22: FCFS ---
\begin{frame}
    \frametitle{1. FCFS (First-Come, First-Serve)}
    \framesubtitle{Paling sederhana tapi paling boros pergerakan}
    \small
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{Konsep FCFS}
            Permintaan dilayani persis sesuai urutan kedatangannya.
        \end{block}
        \textbf{Karakteristik:}
        \begin{itemize}
            \item \tem{Adil}: Tidak ada request yang kelaparan (\textit{starvation}).
            \item \tem{Tidak efisien}: \textit{Head} disk bisa bergerak liar bolak-balik dari ujung ke ujung.
        \end{itemize}

        \column{0.48\textwidth}
        \begin{exampleblock}{Contoh Kasus}
            Antrean request track: 98, 183, 37, 122, 14, 124, 65, 67\\
            Posisi Head Awal: \textbf{53}
            
            \vspace{0.2cm}
            \textit{Pergerakan head:}\\
            53 $\to$ 98 $\to$ 183 $\to$ 37 $\to$ 122 $\to$ 14 $\to$ 124 $\to$ 65 $\to$ 67
            
            \vspace{0.2cm}
            \textbf{Total Pergerakan:} 640 silinder!
        \end{exampleblock}
    \end{columns}
\end{frame}

%--- Slide 23: SSTF ---
\begin{frame}
    \frametitle{2. SSTF (Shortest Seek Time First)}
    \framesubtitle{Pilih yang paling dekat dari posisi sekarang}
    \small
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{Konsep SSTF}
            Melayani permintaan yang posisinya \tem{paling dekat} dengan posisi \textit{head} saat ini, mengabaikan urutan kedatangan.
        \end{block}
        \textbf{Karakteristik:}
        \begin{itemize}
            \item \tem{Efisien}: Mengurangi total pergerakan head secara drastis dibanding FCFS.
            \item \tem{Masalah}: Rentan \textit{\textbf{Starvation}}. Request yang jauh bisa tidak dilayani terus menerus jika request baru bermunculan di dekat head.
        \end{itemize}

        \column{0.48\textwidth}
        \begin{exampleblock}{Contoh Kasus}
            Antrean: 98, 183, 37, 122, 14, 124, 65, 67\\
            Posisi Head Awal: \textbf{53}
            
            \vspace{0.2cm}
            \textit{Pergerakan head (cari terdekat):}\\
            53 $\to$ 65 $\to$ 67 $\to$ 37 $\to$ 14 $\to$ 98 $\to$ 122 $\to$ 124 $\to$ 183
            
            \vspace{0.2cm}
            \textbf{Total Pergerakan:} 236 silinder! (Jauh lebih baik dari 640)
        \end{exampleblock}
    \end{columns}
\end{frame}

%--- Slide 24: SCAN ---
\begin{frame}
    \frametitle{3. Algoritma SCAN (Elevator Algorithm)}
    \framesubtitle{Maju terus pantang mundur (sampai mentok)}
    \small
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{Konsep SCAN}
            \textit{Head} bergerak ke satu arah (misal ke track dalam), melayani semua request di jalurnya. Saat sampai di \tem{ujung disk}, ia berbalik arah dan melayani sisa request di jalur pulang.
        \end{block}
        
        \begin{exampleblock}{Analogi: Lift}
            Lift naik melayani semua lantai yang ditekan sampai lantai teratas, baru setelah itu turun melayani lantai bawah.
        \end{exampleblock}

        \column{0.48\textwidth}
        \textbf{Karakteristik:}
        \begin{itemize}
            \item Mencegah \textit{starvation} ekstrem dari SSTF.
            \item Efisien seperti arah sapuan.
            \item \tem{Masalah}: Kurang adil bagi request yang berada tepat di belakang arah pergerakan \textit{head}, karena harus menunggu \textit{head} sampai ujung dan berbalik arah.
        \end{itemize}
    \end{columns}
\end{frame}

%--- Slide 25: C-SCAN ---
\begin{frame}
    \frametitle{4. Algoritma C-SCAN (Circular SCAN)}
    \framesubtitle{Sapuan satu arah demi keadilan}
    \small
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{Konsep C-SCAN}
            Sama seperti SCAN, namun ketika \textit{head} mencapai ujung disk, ia \tem{langsung melompat kembali ke titik awal} ujung lainnya \textbf{tanpa melayani request} di jalan pulang.
        \end{block}
        
        \textbf{Mengapa C-SCAN lebih baik dari SCAN?}
        \begin{itemize}
            \item Pada SCAN biasa, ketika head berbalik dari ujung, area tersebut baru saja disapu (kemungkinan request barunya sedikit).
            \item Area di ujung lain sudah menunggu sangat lama.
        \end{itemize}

        \column{0.48\textwidth}
        \begin{tikzpicture}[scale=0.85, transform shape]
            \draw[thick, ->] (0,0) -- (6,0) node[right] {\scriptsize Track};
            \draw[thick] (0,-0.1) -- (0,0.1) node[above] {\scriptsize 0};
            \draw[thick] (5,-0.1) -- (5,0.1) node[above] {\scriptsize 199};
            
            % C-SCAN path
            \draw[blue, thick, ->] (1.5, -0.5) -- (5, -0.5) node[midway, below] {\scriptsize Sapu \& Layani};
            \draw[red, thick, dashed, ->] (5, -1.2) -- (0, -1.2) node[midway, below] {\scriptsize Lompat Balik Cepat};
            \draw[blue, thick, ->] (0, -1.9) -- (1.2, -1.9) node[midway, below] {\scriptsize Lanjut Sapu};
        \end{tikzpicture}
        \vspace{0.2cm}
        \begin{itemize}
            \item \tem{Kelebihan}: Memberikan waktu tunggu yang jauh lebih seragam bagi semua request.
        \end{itemize}
    \end{columns}
\end{frame}

%--- Slide 26: LOOK & C-LOOK ---
\begin{frame}
    \frametitle{5. LOOK dan C-LOOK}
    \framesubtitle{Varian pintar dari SCAN}
    \small
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{block}{Masalah SCAN/C-SCAN Murni}
            Mereka selalu bergerak \tem{sampai ujung disk (track 0 atau ujung maksimal)}, bahkan jika tidak ada lagi request di arah tersebut. Ini membuang waktu.
        \end{block}
        
        \textbf{Algoritma LOOK:}
        \begin{itemize}
            \item Varian SCAN. Head hanya bergerak sejauh \tem{request terakhir} di arah tersebut, lalu berbalik.
        \end{itemize}

        \column{0.48\textwidth}
        \textbf{Algoritma C-LOOK:}
        \begin{itemize}
            \item Varian C-SCAN. Head menyapu satu arah, berhenti di request terjauh, lalu \tem{melompat balik ke request terendah} (bukan track 0).
            \item Ini adalah algoritma yang \tem{paling sering diimplementasikan} di dunia nyata untuk HDD mekanis.
        \end{itemize}
    \end{columns}
\end{frame}

%--- Slide 27: Ringkasan Perbandingan ---
\begin{frame}
    \frametitle{Ringkasan Perbandingan Algoritma}
    \framesubtitle{Mana yang terbaik?}
    \small
    \begin{table}[]
        \centering
        \scriptsize
        \begin{tabular}{p{0.18\textwidth} p{0.35\textwidth} p{0.35\textwidth}}
            \toprule
            \textbf{Algoritma} & \textbf{Kelebihan Utama} & \textbf{Kelemahan Utama} \\
            \midrule
            \rowcolor{gray!10}
            \textbf{FCFS} & Adil, sangat mudah diimplementasikan. & Sangat lambat, pergerakan head boros. \\
            \textbf{SSTF} & Kinerja sangat baik untuk batch request. & Resiko \textit{starvation} tinggi bagi track jauh. \\
            \rowcolor{gray!10}
            \textbf{SCAN} & Kinerja baik, mencegah \textit{starvation}. & Tidak adil bagi area yang baru saja dilewati head. \\
            \textbf{C-SCAN} & Waktu tunggu sangat seragam/adil. & Head tetap pergi ke ujung disk membuang waktu. \\
            \rowcolor{gray!10}
            \textbf{C-LOOK} & Efisien, seragam, tidak ke ujung disk. & Overhead komputasi mencari letak request ujung. \\
            \bottomrule
        \end{tabular}
    \end{table}
    
    \vspace{0.2cm}
    \begin{exampleblock}{Catatan untuk SSD}
        SSD (Solid State Drive) \tem{tidak memiliki piringan berputar} atau lengan mekanis (tidak ada \textit{seek time}). OS biasanya menggunakan FCFS yang dimodifikasi, atau membiarkan *firmware* SSD mengatur penjadwalannya sendiri.
    \end{exampleblock}
\end{frame}

%--- Slide 28: Checkpoint 3 ---
\begin{frame}
    \frametitle{Checkpoint 3: Disk Scheduling}
    \framesubtitle{Cek pemahaman sebelum latihan}
    \small
    \begin{enumerate}
        \item Apa dua komponen utama yang membuat operasi disk I/O lambat?
        \item Mengapa algoritma SSTF bisa menyebabkan sebuah proses menunggu selamanya (\textit{starvation})?
        \item Apa bedanya algoritma SCAN biasa dengan LOOK?
        \item Mengapa algoritma penjadwalan pergerakan head (seperti C-SCAN) menjadi kurang relevan pada teknologi penyimpanan modern saat ini?
    \end{enumerate}
    
    \vspace{0.3cm}
    \begin{block}{Kunci Jawaban Singkat}
        \scriptsize
        1. \textit{Seek time} (pergerakan lengan) dan \textit{Rotational latency} (putaran piringan).\\
        2. Jika request baru selalu berdatangan di area dekat head, request di ujung disk tidak akan pernah dipilih.\\
        3. SCAN berjalan mentok sampai ujung disk. LOOK hanya berjalan sejauh posisi request terjauh lalu berbalik.\\
        4. Karena penggunaan \textbf{SSD (Solid State Drive)} yang murni elektronik tanpa lengan penggerak/piringan, sehingga tidak ada konsep fisik \textit{seek time}.
    \end{block}
\end{frame}

\section{Latihan: Perhitungan Disk Scheduling}

%--- Slide 29: Soal Latihan ---
\begin{frame}
    \frametitle{Soal Latihan}
    \framesubtitle{Hitung total pergerakan head (dalam silinder)}
    \small
    \begin{block}{Skenario}
        Sebuah disk drive memiliki \textbf{200 silinder}, dinomori dari \tem{0 hingga 199}.\\
        Posisi \textit{head} disk saat ini berada di silinder \tem{53}.
    \end{block}
    
    Antrean permintaan akses I/O (\textit{queue}) datang dengan urutan berikut:
    \begin{center}
        \Large \textbf{98, 183, 37, 122, 14, 124, 65, 67}
    \end{center}
    
    \vspace{0.3cm}
    \textbf{Tugas:}
    Hitung \textbf{total jarak pergerakan \textit{head}} untuk algoritma berikut:
    \begin{enumerate}
        \item \textbf{FCFS} (First-Come, First-Serve)
        \item \textbf{SSTF} (Shortest Seek Time First)
        \item \textbf{SCAN} (Asumsi arah awal menuju silinder lebih besar / 199)
        \item \textbf{C-SCAN} (Asumsi arah awal menuju silinder lebih besar / 199)
    \end{enumerate}
\end{frame}

        \end{itemize}
        
        \column{0.48\textwidth}
        \textbf{Lanjutan:}
        \begin{itemize}
            \item $|14 - 124| = 110$
            \item $|124 - 65| = 59$
            \item $|65 - 67| = 2$
        \end{itemize}
        
        \vspace{0.3cm}

%--- Slide 32: Jawaban SCAN ---
\begin{frame}
    \frametitle{Jawaban: SCAN}
    \framesubtitle{Maju ke ujung (199), lalu berbalik arah}
    \small
    \textbf{Antrean (Sorted):} 14, 37, 65, 67, 98, 122, 124, 183\\
    \textbf{Posisi Awal:} 53 \quad \textbf{Arah:} Ke atas (menuju 199)
    
    \vspace{0.3cm}
    \textbf{Jalur Pergerakan:}
    53 $\rightarrow$ 65 $\rightarrow$ 67 $\rightarrow$ 98 $\rightarrow$ 122 $\rightarrow$ 124 $\rightarrow$ 183 $\rightarrow$ \tem{199} $\rightarrow$ 37 $\rightarrow$ 14
    
    \vspace{0.3cm}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{Perhitungan Jarak:}
        \begin{itemize}
            \item Perjalanan naik (53 ke 199):\\
                  $|199 - 53| = 146$
            \item Perjalanan turun (199 ke 14):\\
                  $|199 - 14| = 185$
        \end{itemize}
        
        \column{0.48\textwidth}
        \begin{exampleblock}{Cara Cepat}
            Tujuan terjauh ke kanan: 199\\
            Tujuan terjauh ke kiri: 14\\
            Total = (199 - 53) + (199 - 14)\\
            Total = 146 + 185 = \textbf{331 silinder}
        \end{exampleblock}
    \end{columns}
\end{frame}

\end{frame}
