01_thread_basic.c

Thread Abstraction: Membuat & Join Thread

Contoh paling sederhana menggunakan Pthread API. Dua thread berjalan paralel, lalu main menunggu keduanya selesai.

pthread_create pthread_join non-determinism
Perhatikan: urutan baris "Thread A berjalan" dan "Thread B berjalan" bisa berbeda setiap kali dijalankan — inilah yang disebut non-determinism pada program concurrent. 
/*
 * 01_thread_basic.c – Thread Abstraction
 * Compile: gcc -Wall -pthread -o 01_thread_basic 01_thread_basic.c
 */
#include <stdio.h>
#include <pthread.h>

void *my_thread(void *arg) {
    char *name = (char *) arg;
    printf("Thread %s berjalan\n", name);
    return NULL;
}

int main(void) {
    pthread_t t1, t2;

    pthread_create(&t1, NULL, my_thread, "A");
    pthread_create(&t2, NULL, my_thread, "B");

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    printf("Main selesai\n");
    return 0;
}
02_race_condition.c

Race Condition: counter++ yang Tidak Aman

Dua thread masing-masing melakukan 1.000.000 increment pada satu variabel shared tanpa proteksi — hasilnya tidak selalu 2.000.000.

shared variable LOAD–ADD–STORE lost update interleaving
Mengapa < 2.000.000?  counter++ terdiri dari tiga instruksi mesin: LOAD (baca dari memori), ADD (tambah di register), STORE (tulis balik). Jika T1 di-preempt setelah LOAD, T2 menulis nilai baru, lalu T1 menimpa hasil T2 → satu update hilang (lost update). 
/*
 * 02_race_condition.c – Race Condition
 * Compile: gcc -Wall -pthread -O0 -o 02_race_condition 02_race_condition.c
 * Jalankan beberapa kali dan amati nilainya berubah.
 */
#include <stdio.h>
#include <pthread.h>

#define LOOP 1000000

int counter = 0; /* shared variable */

void *increment(void *arg) {
    for (int i = 0; i < LOOP; i++) {
        counter++; /* LOAD, ADD, STORE — TIDAK atomik! */
    }
    return NULL;
}

int main(void) {
    pthread_t t1, t2;

    pthread_create(&t1, NULL, increment, NULL);
    pthread_create(&t2, NULL, increment, NULL);

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    printf("Ekspektasi : %d\n", LOOP * 2);
    printf("Hasil nyata: %d\n", counter);

    if (counter == LOOP * 2)
        printf("STATUS: Kebetulan benar (coba jalankan lagi)\n");
    else
        printf("STATUS: Race condition terdeteksi! Kehilangan %d update.\n",
               LOOP * 2 - counter);

    return 0;
}
03_mutex_fix.c

Critical Section: Perbaikan dengan Mutex

Program yang sama, tapi counter++ dilindungi dengan pthread_mutex_lock / pthread_mutex_unlock. Hasilnya selalu tepat 2.000.000.

mutex critical section mutual exclusion pthread_mutex_lock
Prinsipnya: mutex memastikan hanya satu thread yang mengeksekusi counter++ pada satu waktu → syarat mutual exclusion terpenuhi. Ini adalah teaser materi minggu berikutnya (lock & synchronization primitives). 
/*
 * 03_mutex_fix.c – Mutex sebagai solusi Critical Section
 * Compile: gcc -Wall -pthread -O0 -o 03_mutex_fix 03_mutex_fix.c
 */
#include <stdio.h>
#include <pthread.h>

#define LOOP 1000000

int counter = 0;
pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;

void *increment(void *arg) {
    for (int i = 0; i < LOOP; i++) {
        pthread_mutex_lock(&lock);   /* masuk critical section */
        counter++;                   /* ← protected! */
        pthread_mutex_unlock(&lock); /* keluar critical section */
    }
    return NULL;
}

int main(void) {
    pthread_t t1, t2;

    pthread_create(&t1, NULL, increment, NULL);
    pthread_create(&t2, NULL, increment, NULL);

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    pthread_mutex_destroy(&lock);

    printf("Ekspektasi : %d\n", LOOP * 2);
    printf("Hasil nyata: %d\n", counter);

    if (counter == LOOP * 2)
        printf("STATUS: Benar! Mutex melindungi critical section.\n");

    return 0;
}
04_naive_flag.c

Percobaan Naif: Flag Sederhana (Bisa Hang!)

Upaya manual melindungi critical section dengan dua variabel flag. Pendekatan ini gagal memenuhi syarat Progress dan bisa menyebabkan deadlock.

naive solution progress violation deadlock risk busy-waiting
⚠ Program ini bisa HANG. Jika Thread 0 dan Thread 1 sama-sama men-set flag menjadi 1 hampir bersamaan, keduanya masuk ke loop while dan saling menunggu selamanya. Tekan Ctrl+C untuk keluar. Ini membuktikan bahwa solusi flag naif tidak memenuhi syarat Progress. 
/*
 * 04_naive_flag.c – Solusi Flag Naif (bisa hang!)
 * Compile: gcc -Wall -pthread -O0 -o 04_naive_flag 04_naive_flag.c
 * PERINGATAN: program ini bisa hang. Tekan Ctrl+C untuk keluar.
 */
#include <stdio.h>
#include <pthread.h>

#define LOOP 500000

int counter = 0;
volatile int flag[2] = {0, 0};

void *thread0(void *arg) {
    for (int i = 0; i < LOOP; i++) {
        flag[0] = 1;        /* nyatakan niat masuk */
        while (flag[1]);    /* tunggu kalau thread1 di dalam */
        counter++;
        flag[0] = 0;
    }
    return NULL;
}

void *thread1(void *arg) {
    for (int i = 0; i < LOOP; i++) {
        flag[1] = 1;
        while (flag[0]);    /* masalah: jika keduanya set flag
                               hampir bersamaan → saling tunggu */
        counter++;
        flag[1] = 0;
    }
    return NULL;
}

int main(void) {
    pthread_t t0, t1;

    printf("Memulai... (mungkin hang, tekan Ctrl+C jika perlu)\n");

    pthread_create(&t0, NULL, thread0, NULL);
    pthread_create(&t1, NULL, thread1, NULL);

    pthread_join(t0, NULL);
    pthread_join(t1, NULL);

    printf("Ekspektasi : %d\n", LOOP * 2);
    printf("Hasil nyata: %d\n", counter);

    return 0;
}
